cmake_minimum_required(VERSION 3.24)
# CUDA 12.4 on this Ubuntu workstation supports g++ 13 as its host compiler.
if(NOT DEFINED CMAKE_CUDA_HOST_COMPILER AND EXISTS "/usr/bin/g++-13")
    set(CMAKE_CUDA_HOST_COMPILER "/usr/bin/g++-13" CACHE FILEPATH "CUDA host compiler")
endif()
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES native CACHE STRING "CUDA architectures")
endif()
project(dynamics LANGUAGES C CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

foreach(dependency raylib/src/raylib.h eigen/Eigen/Core imgui/imgui.h rlImGui/rlImGui.h)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/${dependency}")
        message(FATAL_ERROR "Initialize dependencies: git submodule update --init --depth 1")
    endif()
endforeach()
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(external/raylib EXCLUDE_FROM_ALL)
add_library(imgui STATIC
    external/imgui/imgui.cpp external/imgui/imgui_draw.cpp
    external/imgui/imgui_tables.cpp external/imgui/imgui_widgets.cpp
    external/rlImGui/rlImGui.cpp)
target_include_directories(imgui SYSTEM PUBLIC external/imgui external/rlImGui)
target_compile_definitions(imgui PRIVATE NO_FONT_AWESOME)
target_link_libraries(imgui PUBLIC raylib)
find_package(CUDAToolkit REQUIRED)
# GPU implementations use CUDA, while CPU solvers and viewers use C++.
set_source_files_properties(core/xpbd_gpu.cpp core/parallel_cloth_gpu.cpp core/mpm_gpu.cpp
    PROPERTIES LANGUAGE CUDA)
add_library(core STATIC
    utils/cuda_utils.cpp
    core/xpbd.cpp core/pbd.cpp core/pd.cpp
    core/vbd.cpp core/avbd.cpp core/bw.cpp core/xpbd_newton.cpp
    core/mpm.cpp core/parallel_cloth.cpp
    core/xpbd_gpu.cpp core/parallel_cloth_gpu.cpp core/mpm_gpu.cpp)
target_include_directories(core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(core SYSTEM PUBLIC external/eigen external/raylib/src)
target_compile_definitions(core PUBLIC EIGEN_NO_CUDA)
target_link_libraries(core PUBLIC CUDA::cudart)
function(configure_cuda target)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-Xcompiler=-Wall,-Wextra>)
endfunction()
configure_cuda(core)
target_compile_options(core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)

add_library(test_fixtures STATIC EXCLUDE_FROM_ALL tests/scene_fixtures.cpp)
target_link_libraries(test_fixtures PUBLIC core)
target_compile_options(test_fixtures PRIVATE -Wall -Wextra -Wpedantic)

add_executable(pendulum scenes/pendulum.cpp)
target_link_libraries(pendulum PRIVATE core imgui)
target_compile_options(pendulum PRIVATE -Wall -Wextra -Wpedantic)

foreach(scene spring double_pendulum cloth falling_cloth mpm)
    add_executable(${scene} scenes/${scene}.cpp)
    target_link_libraries(${scene} PRIVATE core imgui)
    target_compile_options(${scene} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()

option(DYNAMICS_REFERENCE_TESTS "Build CPU numerical oracles; no CPU viewers" OFF)
option(DYNAMICS_SANITIZE "Sanitize the optional CPU reference tests" OFF)
if(DYNAMICS_SANITIZE)
    set(DYNAMICS_REFERENCE_TESTS ON)
    target_compile_options(test_fixtures PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_compile_options(core PRIVATE
        $<$<COMPILE_LANGUAGE:CXX>:-fsanitize=address,undefined;-fno-omit-frame-pointer>
        $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-fsanitize=address;-Xcompiler=-fsanitize=undefined;-Xcompiler=-fno-omit-frame-pointer>)
    target_link_options(core PUBLIC
        $<$<LINK_LANGUAGE:CXX>:-fsanitize=address,undefined>
        $<$<LINK_LANGUAGE:CUDA>:-Xcompiler=-fsanitize=address;-Xcompiler=-fsanitize=undefined>)
endif()
if(BUILD_TESTING AND DYNAMICS_REFERENCE_TESTS)
    foreach(target solver_tests reference_tests parallel_cloth_tests)
        add_executable(${target} ${target}.cpp)
        target_link_libraries(${target} PRIVATE test_fixtures)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(DYNAMICS_SANITIZE)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
        add_test(NAME reference_${target} COMMAND ${target})
    endforeach()
endif()
